
#include<stdio.h>
int main()
{
int n,bt[10],rt[10],p[10],wt[10],tat[10],i,time=0,complete=0,highest;
printf("Enter number of processes: ");
scanf("%d",&n);
if(n<1||n>10)
{
printf("Invalid number of processes\n");
return 0;
}
for(i=0;i<n;i++)
{
printf("Enter Burst Time and Priority of P%d: ",i+1);
scanf("%d%d",&bt[i],&p[i]);
rt[i]=bt[i];
}
printf("\nExecution Order:\n");
while(complete<n)
{
highest=-1;
for(i=0;i<n;i++)
{
if(rt[i]>0&&(highest==-1||p[i]<p[highest]))
highest=i;
}
printf("P%d ",highest+1);
rt[highest]--;
time++;
if(rt[highest]==0)
{
tat[highest]=time;
wt[highest]=tat[highest]-bt[highest];
complete++;
}
}
printf("\n\nPID\tPriority\tBT\tWT\tTAT\n");
for(i=0;i<n;i++)
printf("P%d\t%d\t\t%d\t%d\t%d\n",i+1,p[i],bt[i],wt[i],tat[i]);
return 0;
}
