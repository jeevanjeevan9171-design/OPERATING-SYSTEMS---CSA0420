#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/stat.h>
#include <dirent.h>

int main() {
    int fd, flags;
    char buf[50];
    off_t pos;
    struct stat st;
    DIR *dp;
    struct dirent *dir;

    fd = open("test.txt", O_CREAT | O_RDWR, 0644);
    if(fd < 0) {
        printf("File opening failed\n");
        return 1;
    }
    write(fd, "Operating Systems Lab", 21);

    pos = lseek(fd, 0, SEEK_SET);
    printf("lseek: file offset set to %ld\n", (long)pos);

    read(fd, buf, 21);
    buf[21] = '\0';
    printf("read after lseek: %s\n", buf);

    flags = fcntl(fd, F_GETFL);
    printf("fcntl: file status flags = %d\n", flags);
    fcntl(fd, F_SETFL, flags | O_APPEND);
    printf("fcntl: O_APPEND flag set\n");
    close(fd);

    if(stat("test.txt", &st) == 0) {
        printf("stat: file size = %ld bytes\n", (long)st.st_size);
        printf("stat: file permissions = %o\n", st.st_mode & 0777);
        printf("stat: number of hard links = %ld\n", (long)st.st_nlink);
    }

    dp = opendir(".");
    if(dp == NULL) {
        printf("Directory opening failed\n");
        return 1;
    }
    printf("opendir + readdir: files in current directory\n");
    while((dir = readdir(dp)) != NULL) {
        if(dir->d_name[0] != '.')
            printf("%s\n", dir->d_name);
    }
    closedir(dp);

    return 0;
}
