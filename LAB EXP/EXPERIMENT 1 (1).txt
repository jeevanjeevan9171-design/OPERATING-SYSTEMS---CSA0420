
#include<stdio.h>
#include<windows.h>
int main()
{
STARTUPINFO si={0};
PROCESS_INFORMATION pi={0};
si.cb=sizeof(si);
printf("Parent Process ID: %lu\n",(unsigned long)GetCurrentProcessId());
if(CreateProcess(NULL,"notepad.exe",NULL,NULL,FALSE,0,NULL,NULL,&si,&pi))
{
printf("Child Process ID: %lu\n",(unsigned long)pi.dwProcessId);
printf("Parent Process ID: %lu\n",(unsigned long)GetCurrentProcessId());
CloseHandle(pi.hProcess);
CloseHandle(pi.hThread);
}
else
{
printf("Process creation failed!\n");
}
return 0;
}

}
