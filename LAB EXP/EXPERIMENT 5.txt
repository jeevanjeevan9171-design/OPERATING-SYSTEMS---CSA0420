
#include<stdio.h>
int main()
{
int n,i,j,bt[10],p[10],pid[10],wt[10],tat[10],temp;
printf("Enter number of processes: ");
scanf("%d",&n);
if(n<1||n>10)
{
printf("Enter processes between 1 and 10\n");
return 0;
}
for(i=0;i<n;i++)
{
pid[i]=i+1;
printf("Enter Burst Time and Priority of P%d: ",i+1);
scanf("%d%d",&bt[i],&p[i]);
}
for(i=0;i<n-1;i++)
{
for(j=i+1;j<n;j++)
{
if(p[i]>p[j])
{
temp=p[i];p[i]=p[j];p[j]=temp;
temp=bt[i];bt[i]=bt[j];bt[j]=temp;
temp=pid[i];pid[i]=pid[j];pid[j]=temp;
}
}
}
wt[0]=0;
for(i=1;i<n;i++)
wt[i]=wt[i-1]+bt[i-1];
for(i=0;i<n;i++)
tat[i]=wt[i]+bt[i];
printf("\nPID\tPriority\tBT\tWT\tTAT\n");
for(i=0;i<n;i++)
printf("P%d\t%d\t\t%d\t%d\t%d\n",pid[i],p[i],bt[i],wt[i],tat[i]);
getchar();
getchar();
return 0;
}
